# 1. 유저 모델 가져오기
from django.contrib.auth import get_user_model
User = get_user_model()

# 2. 비밀번호를 바꿀 유저 찾기 (예: 이메일이 admin@admin.com인 유저)
user = User.objects.get(email='d@d.com')

# 3. 비밀번호 강제 세팅 (여기에 원하는 새 비밀번호 입력)
user.set_password('1')

# 4. 데이터베이스에 저장!
user.save()

# 5. 쉘 종료
exit()
